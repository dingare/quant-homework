### Verified user submission

```python
def find_root(x,root):
    while root[x]!=x:
        root[x] = root[root[x]]
        x = root[x]
    return x

def connected_pairs(n, edges):
    res = []
    size = [1]*n
    count = 0
    root = [i for i in range(n)]
    for u, v in edges:
        ru = find_root(u,root)
        rv = find_root(v,root)
        if ru!=rv:
            if size[ru]<size[rv]:
                u,v = v,u
                ru,rv = rv,ru
            count += size[ru]*size[rv]
            root[rv] = ru
            size[ru] += size[rv]
        res.append(count)
    return (res)
```
